   /**
    * Returns true if the path is inside a Wine/Proton prefix.
    * O(1) string check per D-04.
    */
   function isWinePrefixPath(filePath: string): boolean {
     return process.platform === "linux"
       && filePath.includes("/compatdata/")
       && filePath.includes("/pfx/");
   }

   /**
    * For Wine prefix paths on Linux, resolve on-disk casing of the leaf
    * file/directory. No dirCache per D-09 — this is a safety net for
    * scattered individual calls, not a bulk deployment loop.
    */
   async function resolveCaseIfWinePrefix(absPath: string): Promise<string> {
     if (!isWinePrefixPath(absPath)) {
       return absPath;
     }
     const dir = path.dirname(absPath);
     const base = path.basename(absPath);
     return resolvePathCase(dir, base);
   }
   ```

   import { readdirSync } from "original-fs";  // already in the re-export, but need it for sync resolution

   /**
    * Synchronous case-folding for watch paths on Wine prefixes.
    * Uses readdirSync because fs.watch is synchronous.
    */
   function resolvePathCaseSync(absPath: string): string {
     if (!isWinePrefixPath(absPath)) {
       return absPath;
     }
     // Resolve parent directory segments + leaf synchronously
     const segments = absPath.split("/");
     let current = "/";
     for (let i = 1; i < segments.length; i++) {
       if (segments[i] === "") continue;
       try {
         const entries = readdirSync(current);
         const lower = segments[i].toLowerCase();
         const match = entries.find((e) => e.toLowerCase() === lower);
         current = path.join(current, match !== undefined ? match : segments[i]);
       } catch {
         // Can't read directory — keep remaining segments as-is
         current = path.join(current, ...segments.slice(i));
         break;
       }
     }
     return current;
   }

   function watch(
     filename: string,
     ...args: any[]
   ): fs.FSWatcher {
     return watchOriginal(resolvePathCaseSync(filename), ...args);
   }
   ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Wine prefix path detection | String check determines whether case-folding is applied; false positive = unnecessary readdir overhead; false negative = original bug persists |
| Filesystem readdir in shim | resolvePathCase reads directory listings to find case matches; attacker-controlled filenames could theoretically cause confusion |
| Synchronous watch wrapper | resolvePathCaseSync uses readdirSync which blocks the event loop; only triggers on Wine prefix paths which are rare |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-01 | Tampering | isWinePrefixPath | accept | String check is advisory; false positives only add readdir overhead, no security impact. False negatives preserve pre-shim behavior (no regression). |
| T-14-02 | Denial of Service | resolvePathCaseSync | mitigate | Sync readdir only fires for watch() on Wine prefix paths (rare, typically 1-2 calls per game). Not in hot path. Errors caught and original path preserved. |
| T-14-03 | Information Disclosure | resolvePathCase readdir | accept | readdir results are directory listings the calling process already has access to. No privilege escalation. |
| T-14-04 | Elevation of Privilege | fs.ts shim | accept | Shim does not change permissions or bypass access controls. It only resolves casing before passing to the original fs function which enforces normal POSIX permissions. |
</threat_model>
