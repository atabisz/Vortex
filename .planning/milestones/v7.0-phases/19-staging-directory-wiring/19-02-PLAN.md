export async function suggestStagingPath(
  api: IExtensionApi,
  gameId: string,
): Promise<string> {
  const state = api.getState();
  const modPaths = modPathsForGame(state, gameId);
  let statModPath: fs.Stats;

  const idModPath = async (testPath: string) => {
    try {
      statModPath = await fs.statAsync(testPath);
    } catch (err) {
      const code = getErrorCode(err);
      if (code === "ENOENT") {
        await idModPath(path.dirname(testPath));
      } else {
        throw err;
      }
    }
  };

  await idModPath(modPaths[""]);
  const statUserData = await fs.statAsync(getVortexPath("userData"));

  let suggestion: string;
  if (statModPath.dev === statUserData.dev || process.platform !== "win32") {
    suggestion = path.join("{USERDATA}", "{game}", "mods");
  } else {
    const volume = winapi.GetVolumePathName(modPaths[""]);
    suggestion = path.join(
      volume,
      state.settings.mods.suggestInstallPathDirectory,
      "{game}",
    );
  }
  return suggestion;
}

private suggestPath = async () => {
  const { modPaths, onShowError, suggestInstallPathDirectory } = this.props;
  try {
    const [modPathStats, userDataStats] = await Promise.all([

Per RESEARCH.md Pitfall 2 and Open Question 1: `Settings.tsx:suggestPath()` has its own independent `winapi.GetVolumePathName(modPaths[""])` call at line 1156 that would crash on Linux. This is a separate code path from `discovery.ts:suggestStagingPath()` -- it is the handler for the "Suggest" button in the Settings panel, reachable by the user on Linux.

The existing code stats `path.parse(modPaths[""]).root` to get `modPathStats`. On Linux, `path.parse("/mnt/games/skyrim").root` is always `"/"`, so `modPathStats.dev` is the device ID of the root filesystem -- NOT the device of the actual mod path. This means the `modPathStats.dev === userDataStats.dev` comparison works correctly only on single-device systems (where everything is on root). On multi-device Linux systems, `modPathStats` (stat of `/`) may have the same `dev` as userData (also on root device), causing the same-device branch to be taken even when the game is on a different mount. This is actually safe behavior (falls back to USERDATA), but for correct device-aware suggestions on multi-device Linux, we need the device ID of `modPaths[""]` itself.

Replace lines 1152-1158 (the `let suggestion` through closing brace of the if/else) with a three-branch structure:
