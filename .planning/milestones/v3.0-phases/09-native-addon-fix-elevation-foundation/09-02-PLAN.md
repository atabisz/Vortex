type SpawnerFn = (cmd: string, args: string[]) => ChildProcess;

  _spawner = fn;

function getSpawner(): SpawnerFn {

**Step 2: Add Linux branch in runElevated** at line ~174, inside the `fs.write` callback, AFTER the `fs.closeSync(fd)` try/catch block and BEFORE the `try { winapi.ShellExecuteEx(...) }` block:

```typescript
          if (process.platform === "linux") {
            const proc = getSpawner()("pkexec", [
              process.execPath,
              "--run",
              tmpPath,
            ]);
            proc.on("close", (code: number | null) => {
              if (code === 126) {
                reject(new UserCanceled());
              } else if (code !== null && code !== 0) {
                reject(
                  new Error(`pkexec exited with code ${code}`),
                );
              }
              // code 0 or null: normal exit; IPC handles results
            });
            return resolve(tmpPath);
          }
```

Key details:
- `return resolve(tmpPath)` is called IMMEDIATELY after spawn (not inside 'close' handler) — matches Windows semantics where ShellExecuteEx is fire-and-forget
- The 'close' handler ONLY rejects on error/cancel — success is handled via IPC by callers
- `code === 126` maps to UserCanceled (user dismissed polkit dialog)
- `code === 127` or any other non-zero maps to descriptive Error
- The existing `try { winapi.ShellExecuteEx(...) }` block remains untouched below (it runs only on non-Linux)
