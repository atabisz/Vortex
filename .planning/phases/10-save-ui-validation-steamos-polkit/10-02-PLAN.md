**Part A: Add isSteamOS() to elevated.ts (per D-06, D-08)**

Add at module level, near the top of elevated.ts (after imports):

```typescript
import * as fsNode from "fs";

let _isSteamOS: boolean | undefined;

function isSteamOS(): boolean {
  if (_isSteamOS !== undefined) {
    return _isSteamOS;
  }
  try {
    const content = fsNode.readFileSync("/etc/os-release", "utf8");
    _isSteamOS =
      /^ID=steamos$/im.test(content) ||
      /^ID_LIKE=.*steamos.*$/im.test(content);
  } catch {
    _isSteamOS = false;
  }
  return _isSteamOS;
}
```

**Part B: Add SteamOS sudo -n fallback branch (per D-07, D-08, D-09)**

Replace the current Linux block at lines 189-206 with:

```typescript
if (process.platform === "linux") {
  if (isSteamOS()) {
    // SteamOS: pkexec hangs without polkit agent in Game Mode.
    // Attempt sudo -n (non-interactive) instead.
    const proc = getSpawner()("sudo", [
      "-n",
      process.execPath,
      "--run",
      tmpPath,
    ]);
    proc.on("close", (code: number | null) => {
      if (code !== null && code !== 0) {
        // sudo -n failed (password required or ENOENT)
        const err = new UserCanceled();
        (err as any).message =
          "Elevation is not available in Steam Game Mode. " +
          "Switch to Desktop Mode to perform this operation.";
        reject(err);
      }
      // code 0 or null: normal exit; IPC handles results
    });
    proc.on("error", (spawnErr: Error) => {
      // sudo not found on PATH
      const err = new UserCanceled();
      (err as any).message =
        "Elevation is not available in Steam Game Mode. " +
        "Switch to Desktop Mode to perform this operation.";
      reject(err);
    });
  } else {
    // Standard desktop Linux: use pkexec (unchanged from Phase 9)
    const proc = getSpawner()("pkexec", [
      process.execPath,
      "--run",
      tmpPath,
    ]);
    proc.on("close", (code: number | null) => {
      if (code === 126) {
        reject(new UserCanceled());
      } else if (code !== null && code !== 0) {
        reject(new Error(`pkexec exited with code ${code}`));
      }
      // code 0 or null: normal exit; IPC handles results
    });
  }
  return resolve(tmpPath);
}
```

CRITICAL: The `return resolve(tmpPath)` MUST remain after both branches. The IPC server socket is already listening BEFORE this code block (started earlier in the function). Do NOT move the socket setup. The `sudo -n` spawn uses the same `tmpPath` and relies on the same IPC channel as pkexec.
