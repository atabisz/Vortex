<objective>
Add a Linux empty-state block in NoGameDashlet that guides users when no Steam games
are detected, and add a one-shot Steam detection retry in GameModeManager to handle
the case where Vortex launches before Steam finishes loading.

Purpose: Without these changes, a Linux user who opens Vortex on a fresh install sees
a blank screen when no games are found, with no guidance on what to do. And if Vortex
starts before Steam finishes loading, games are never detected unless the user restarts
Vortex.

Output: Modified NoGameDashlet.tsx with empty-state UI, modified GameModeManager.ts
with one-shot retry, and corresponding unit tests.
</objective>

   ```typescript
   public startQuickDiscovery(games?: IGame[]) {
     return this.reloadStoreGames()
       .then(() =>
         quickDiscovery(
           games ?? this.mKnownGames,
           this.mStore.getState().settings.gameMode.discovered,
           this.onDiscoveredGame,
           this.onDiscoveredTool,
         ),
       )
       .then((result) => {
         this.postDiscovery();

         // D-09: One-shot retry on Linux when no games detected
         // Handles race where Vortex starts before Steam finishes loading
         if (process.platform === "linux") {
           const discovered = this.mStore.getState().settings.gameMode.discovered;
           const hasGames = Object.keys(discovered).some(
             (gameId) => discovered[gameId].path !== undefined,
           );
           if (!hasGames) {
             log("debug", "No games found on Linux, retrying after delay");
             Bluebird.delay(2000)
               .then(() => this.reloadStoreGames())
               .then(() =>
                 quickDiscovery(
                   games ?? this.mKnownGames,
                   this.mStore.getState().settings.gameMode.discovered,
                   this.onDiscoveredGame,
                   this.onDiscoveredTool,
                 ),
               )
               .then(() => {
                 this.postDiscovery();
               })
               .catch((err) => {
                 log("debug", "Steam retry failed", err);
               });
           }
         }

         return result;
       });
   }
   ```
