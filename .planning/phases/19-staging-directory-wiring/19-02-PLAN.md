```typescript
export async function suggestStagingPath(
  api: IExtensionApi,
  gameId: string,
): Promise<string> {
  const state = api.getState();
  const modPaths = modPathsForGame(state, gameId);
  let statModPath: fs.Stats;

  const idModPath = async (testPath: string) => {
    try {
      statModPath = await fs.statAsync(testPath);
    } catch (err) {
      const code = getErrorCode(err);
      if (code === "ENOENT") {
        await idModPath(path.dirname(testPath));
      } else {
        throw err;
      }
    }
  };

  await idModPath(modPaths[""]);
  const statUserData = await fs.statAsync(getVortexPath("userData"));

  let suggestion: string;
  if (statModPath.dev === statUserData.dev || process.platform !== "win32") {
    suggestion = path.join("{USERDATA}", "{game}", "mods");
  } else {
    const volume = winapi.GetVolumePathName(modPaths[""]);
    suggestion = path.join(
      volume,
      state.settings.mods.suggestInstallPathDirectory,
      "{game}",
    );
  }
  return suggestion;
}
```

```typescript
private suggestPath = async () => {
  const { modPaths, onShowError, suggestInstallPathDirectory } = this.props;
  try {
    const [modPathStats, userDataStats] = await Promise.all([
      fs.statAsync(path.parse(modPaths[""]).root),
      window.api.app
        .getPath("userData")
        .then((userDataPath) => fs.statAsync(userDataPath)),
    ]);

    let suggestion: string;
    if (modPathStats.dev === userDataStats.dev) {
      suggestion = path.join("{USERDATA}", "{game}", "mods");
    } else {
      const volume = winapi.GetVolumePathName(modPaths[""]);
      suggestion = path.join(volume, suggestInstallPathDirectory, "{game}");
    }
    this.changePath(suggestion);
  } catch (err) {
    if (err instanceof UserCanceled) {
      return;
    }
    onShowError("Failed to suggest path", err);
  }
};
```

Key implementation notes:
- `modPathStats` is the stat of `path.parse(modPaths[""]).root` (already computed on line 1146). On Linux, `path.parse("/mnt/games/skyrim").root` is `"/"`, so `modPathStats.dev` equals the device of the root. However, the mountpoint walk starting from `modPaths[""]` itself will correctly find the device boundary by comparing `parentStat.dev` against `modPathStats.dev`.
- Actually, wait -- `modPathStats` stats the volume root, not `modPaths[""]` itself. On Linux the root is always `/`. The `dev` of `/` may differ from the `dev` of `/mnt/games`. So we need to stat `modPaths[""]` (or its nearest existing ancestor) for the walk baseline. However, the existing code already uses `modPathStats.dev === userDataStats.dev` for the same-device check, and on a single-device Linux system, `/` has the same dev as everything on it, so the check works correctly.
- For the walk, use `modPathStats.dev` as the baseline since it's the device at the volume root where the mod path resides. On Linux with a single device, `modPathStats.dev` (stat of `/`) matches all paths, so the `if` branch is taken (correct). On multi-device Linux, `modPathStats.dev` (stat of `/`) differs from userData's dev only if userData is on a different mount -- but the mount walk needs the dev of the actual mod path, not `/`.
- CORRECTION: The walk baseline must be the device of `modPaths[""]` itself, not its root. Add a line to stat `modPaths[""]` (or walk up to find an existing ancestor) before the walk:
